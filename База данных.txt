
DROP TABLE IF EXISTS visits CASCADE;
DROP TABLE IF EXISTS venues CASCADE;
DROP TABLE IF EXISTS clients CASCADE;

CREATE TABLE clients (
    client_id SERIAL PRIMARY KEY,
    full_name VARCHAR(150) NOT NULL,
    phone VARCHAR(20) UNIQUE NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    loyalty_tier VARCHAR(20) DEFAULT 'Basic',
    registration_date DATE NOT NULL,
    share_history_allowed BOOLEAN DEFAULT TRUE
);

CREATE TABLE venues (
    venue_id SERIAL PRIMARY KEY,
    venue_name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL
);

CREATE TABLE visits (
    visit_id SERIAL PRIMARY KEY,
    receipt_number VARCHAR(20) UNIQUE NOT NULL,
    client_id INT NOT NULL REFERENCES clients(client_id) ON DELETE CASCADE,
    venue_id INT NOT NULL REFERENCES venues(venue_id) ON DELETE CASCADE,
    visit_timestamp TIMESTAMP NOT NULL,
    duration_minutes INT,
    amount_spent NUMERIC(10, 2) NOT NULL DEFAULT 0
);

INSERT INTO visits (receipt_number, client_id, venue_id, visit_timestamp, duration_minutes, amount_spent)
SELECT 
    'R-' || LPAD(c.client_id::text, 3, '0') || '-' || v.idx,
    c.client_id,
    FLOOR(RANDOM() * 27 + 1)::int, -- Случайное заведение (теперь от 1 до 27!)
    NOW() - (FLOOR(RANDOM() * 60) || ' days')::interval - (FLOOR(RANDOM() * 24) || ' hours')::interval, -- Случайная дата
    FLOOR(RANDOM() * 180 + 30)::int, -- Длительность (30-210 минут)
    FLOOR(RANDOM() * 4500 + 500)::numeric -- Сумма (500-5000)
FROM clients c
CROSS JOIN (VALUES (1), (2), (3), (4)) AS v(idx);
